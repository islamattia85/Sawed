import { test } from '@playwright/test';
import { isolate } from './support.js';
import fs from 'fs';
const D = '/tmp/claude-0/-home-user-Sawed/74c17acb-5048-514f-a9cb-512d72f36e1c/scratchpad/deep/';
const meta = JSON.parse(fs.readFileSync(D + 'meta.json', 'utf8'));
const truth = JSON.parse(fs.readFileSync(D + 'truth.json', 'utf8'));
const fresh = async (page) => { await isolate(page); await page.goto('/?fresh'); await page.waitForFunction(() => window.__bootSettled === true); };
const ranked = (page) => page.evaluate(() => window.getRecommendation().ranked.map((x) => ({ id: x.plan.id, net: x.net })));
const ANS = {
  heatpump: { heat: 'heatpump' }, storage: { heat: 'storage' }, night: { night: 'immersion' },
  ev_night: { ev: 'have', evtime: 'night' }, ev_evening: { ev: 'have', evtime: 'evening' }, rural: { area: 'rural' }, legacy24: { meter: '24hr' },
  solar: { solar: 'have', batt: 0 }, solar_batt: { solar: 'have', batt: 10 },
};
for (const h of Object.keys(meta)) {
  test('typed ' + h, async ({ page }) => {
    await fresh(page);
    const a = ANS[h] || {};
    const r = await page.evaluate(([h, a, kwh, bill]) => {
      startFlow('full');
      if (a.solar) flowAnswer('bill', Math.round(bill)); else flowAnswer('bill', 'kwh:' + kwh);
      flowAnswer('meter', a.meter || 'smart'); flowAnswer('area', a.area || 'urban');
      flowAnswer('plan', a.meter === '24hr' ? 'EI-ES' : 'EI-24'); flowAnswer('disc', 0);
      flowAnswer('heat', a.heat || 'gas'); if (!a.heat) flowAnswer('night', a.night || 'no');
      if (window.state._flow && !('gas' in window.state._flow) && (window.__flowSteps ? window.__flowSteps() : []).includes('gas')) flowAnswer('gas', 'no');
      if (a.solar){ flowAnswer('solar', 'have'); flowAnswer('roof', 'S'); flowAnswer('tilt', 35); flowAnswer('panels', 9); flowAnswer('battery', a.batt); flowAnswer('price', 0); flowAnswer('grant', 'no'); }
      else flowAnswer('solar', 'no');
      if (a.ev){ flowAnswer('ev', 'have'); flowAnswer('km', 15000); flowAnswer('car', 17); flowAnswer('evtime', a.evtime); } else flowAnswer('ev', 'no');
      flowFinish();
      return { kwp: window.totalKwp ? window.totalKwp() : null };
    }, [h, a, meta[h].import, (truth[h]['EI-24'] || truth[h]['EI-ES']) / 6]);
    fs.writeFileSync(D + `pk_${h}_kwh.json`, JSON.stringify({ info: r, ranked: await ranked(page) }));
  });
  if (h !== 'legacy24') test('file ' + h, async ({ page }) => {
    await fresh(page);
    await page.getByRole('button', { name: /Am I paying too much/ }).click();
    await page.locator('.fl-upload input[type=file]').setInputFiles(D + h + '.csv');
    await page.getByRole('button', { name: /Use this data/ }).click();
    await page.waitForTimeout(1200);
    if (h === 'rural') await page.evaluate(() => { flowAnswer('meter', 'smart'); flowAnswer('area', 'rural'); });
    const info = await page.evaluate(() => ({ kwh: Object.values(window.state.bills).reduce((a, b) => a + b, 0), solar: window.state.has_solar }));
    fs.writeFileSync(D + `pk_${h}_file.json`, JSON.stringify({ info, ranked: await ranked(page) }));
  });
}
