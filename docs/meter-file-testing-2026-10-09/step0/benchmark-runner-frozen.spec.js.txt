/** Answer whatever setup asks, from the home's facts, until the answer shows. */
async function answerAll(page, a, log) {
  for (let i = 0; i < 40; i++) {
    const q = await page.evaluate(() => { const o = document.querySelector('.fl-q [data-q]'); if (o) return o.dataset.q; const own = document.querySelector('.fl-q [id^=flow-own-]'); if (own) return own.id.slice(9); if (document.querySelector('.fl-sups, .fl-sup')) return 'plan'; if (document.querySelector('#flow-bill')) return 'bill'; return null; });
    if (!q) break;
    const v = ({ meter: a.meter || 'smart', area: a.area || 'urban', plan: a.plan || 'EI-24', disc: 0, heat: a.heat || 'gas', heattime: a.heattime || 'day',
      night: a.night || 'no', gas: 'no', gasbill: 0, solar: a.solar || 'no', where: 'east', house: 'semi', roof: 'S', tilt: 35, system: 'custom', panels: a.panels || 9,
      battery: a.battery ?? 0, gridnow: 'no', price: 0, grant: 'no', ev: a.ev || 'no', km: 15000, car: 17, evtime: a.evtime || 'night',
      billwhen: a.billwhen || 'after', billmonths: 'average', filewhen: a.filewhen || 'allyear', bill: a.bill })[q];
    if (v === undefined) { log.push('unanswered ' + q); break; }
    log.push(q + '=' + v);
    await page.evaluate(([q, v]) => window.flowAnswer(q, v), [q, v]);
    await page.waitForTimeout(120);
  }
  await page.evaluate(() => { if (window.state.current_screen === 'flow') window.flowFinish(); });
}
